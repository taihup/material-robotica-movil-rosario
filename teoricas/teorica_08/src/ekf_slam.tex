\begin{frame}
    \frametitle{Definición del Problema de SLAM}
    \note{Información extraída de https://www.youtube.com/watch?v=X30sEgIws0g}
    \note{Información extraída de https://www.ipb.uni-bonn.de/html/teaching/photo12-2021/2021-pho2-16-ekf-slam.pptx.pdf}

    Dado:
    \begin{itemize}
        \item Comandos de control enviados
        
        \begin{equation*}
            \controlCommand_{1:T} = \{\controlCommand_1, \controlCommand_2, \ldots, \controlCommand_T\}
        \end{equation*}
        \item Observaciones
        \begin{equation*}
            \observation_{1:T} = \{\observation_1, \observation_2, \ldots, \observation_T\}
        \end{equation*}
    \end{itemize}

    Se busca:
    \begin{itemize}
        \item Mapa del entorno: $\map$
        \item Trayectoria (o pose actual) del vehículo:
        \begin{equation*}
            \state_{0:T} = \{\state_0, \state_1, \ldots, \state_T\} \qquad \text{o} \qquad \state_{t}
        \end{equation*}
    \end{itemize}
\end{frame}

    \begin{itemize}
        \item Filtro recursivo con pasos de \textbf{predicción} y \textbf{corrección}
        \item Estima: $p(x_t, \map | \observation_{1:t}, \controlCommand_{1:t})$
        \item El Filtro de Kalman es un Filtro de Bayes recursivo para el \textbf{caso lineal Gaussiano}
        \item \textbf{EKF} para lidiar con \textbf{no linealidades}
    \end{itemize}

\begin{frame}
    \frametitle{Algoritmo de Filtro de Kalman Extendido}
    \note{Información extraída de https://www.youtube.com/watch?v=X30sEgIws0g}
    \note{Información extraída de https://www.ipb.uni-bonn.de/html/teaching/photo12-2021/2021-pho2-16-ekf-slam.pptx.pdf}
    
    \begin{algorithmic}[1]
    \Procedure{ExtendedKalmanFilter}{$\mu_{t-1}, \covariance_{t-1}, \controlCommand_{t}, \observation_{t}$}
        \State $\overline{\mu}_{t} = \motionModelFunction{\controlCommand_{t}, \mu_{t-1}}$
        \State $\overline{\covariance}_{t} = \motionModelJacobian_{t} \covariance_{t-1} \motionModelJacobian_{t}^{\top}+\motionParametersCovariance_{t}$
        \Statex
        \State $\kalmanGain_{t} = \overline{\covariance}_{t} \observationModelJacobian_{t}^{\top} (\observationModelJacobian_{t} \overline{\covariance}_{t}  \observationModelJacobian_{t}^{\top} + \observationModelCovariance_{t})^{-1} $
        \State $\mu_{t} = \overline{\mu}_{t} + \kalmanGain_{t} (\observation_{t} - \observationModelFunction{\overline{\mu}_{t}})$
        \State $\covariance_{t} =  (I - \kalmanGain_{t} \observationModelJacobian_{t}) \overline{\covariance}_{t}$
        \State \Return $\mu_{t}, \covariance_{t}$
    \EndProcedure
    \end{algorithmic}
\end{frame}

\begin{frame}
    \frametitle{Inicialización}
    \note{Información extraída de https://www.youtube.com/watch?v=X30sEgIws0g}
    \note{Información extraída de https://www.ipb.uni-bonn.de/html/teaching/photo12-2021/2021-pho2-16-ekf-slam.pptx.pdf}

    \begin{itemize}
        \item La plataforma comienza en su propio sistema de referencia
        \item Todos los landmarks son inicialmente desconocidos (aún no observados)
        \item Dimensión del vector de estado: $2N + 3$ \\
        (3 para la pose del robot + $2N$ para $N$ landmarks en el espacio 2D)
        \begin{equation*}
            \mu_0 =
            \begin{bmatrix}
                0 & 0 & 0 & \dots & 0    
            \end{bmatrix}^{\top}
        \end{equation*}
    \end{itemize}
    
    \begin{equation*}
        \Sigma_0 =
        \begin{bmatrix}
            \Sigma_{x_0} & 0 & \cdots & 0 \\
            0 & \infty I_{2} & \cdots & 0 \\
            \vdots & \vdots & \ddots & \vdots \\
            0 & 0 & \cdots & \infty I_{2}
        \end{bmatrix}
    \end{equation*}
    
    \begin{itemize}
        \item $\Sigma_{x_0}$: Incertidumbre inicial de la pose del robot (típicamente una matriz nula o pequeña)
        \item $\infty I_{2}$: Incertidumbre infinita para los landmarks no observados (en la práctica no se usa infinito para evitar problemas numéricos)
    \end{itemize}

    \note{Podemos limitar el uso de landmarks mediante su índice. Cada vez que se observa un nuevo landmark incrementamos el índice, de modo que ese landmark se incorpora al estado y a la matriz de covarianza. Hacemos esto hasta alcanzar el máximo de landmarks con los que podemos trabajar.}

\begin{frame}
    \frametitle{Actualización de la Covarianza}
    \note{Información extraída de https://www.youtube.com/watch?v=X30sEgIws0g}
    \note{Información extraída de https://www.ipb.uni-bonn.de/html/teaching/photo12-2021/2021-pho2-16-ekf-slam.pptx.pdf}

    \begin{itemize}
        \item La función $g$ solo afecta al movimiento y no a los landmarks
    \end{itemize}

    Jacobiano:
    \begin{equation*}
        G_t = 
        \begin{bmatrix}
            G_{t}^{\state} & 0 \\
            0 & I
        \end{bmatrix}
    \end{equation*}
    donde $G_{t}^{\state}$ es el Jacobiano $3 \times 3$ del movimiento e $I$ es la matriz identidad de $2N \times 2N$.


    Notar que la submatriz identidad $I$ implica que las covarianzas de los landmarks no cambian (se mantienen iguales a la covarianza anterior).

\end{frame}

\begin{frame}
    \frametitle{Jacobiano del Movimiento}
    \note{Información extraída de https://www.youtube.com/watch?v=X30sEgIws0g}
    \note{Información extraída de https://www.ipb.uni-bonn.de/html/teaching/photo12-2021/2021-pho2-16-ekf-slam.pptx.pdf}
    
    La matriz Jacobiana $G_{t}^{\state}$ del modelo de movimiento se calcula como:
    
    \begin{align*}
        G_{t}^{\state} &= \frac{\partial}{\partial(x, y, \theta)^{\top}} \left(
        \begin{bmatrix}
        x \\
        y \\
        \theta
        \end{bmatrix}
        +
        \begin{bmatrix}
        -\frac{v_t}{\omega_t} \sin \theta + \frac{v_t}{\omega_t} \sin(\theta + \omega_t \Delta t) \\
        \frac{v_t}{\omega_t} \cos \theta - \frac{v_t}{\omega_t} \cos(\theta + \omega_t \Delta t) \\
        \omega_t \Delta t
        \end{bmatrix}
        \right) \\
        &= I + \frac{\partial}{\partial(x, y, \theta)^{\top}} 
        \begin{bmatrix}
        -\frac{v_t}{\omega_t} \sin \theta + \frac{v_t}{\omega_t} \sin(\theta + \omega_t \Delta t) \\
        \frac{v_t}{\omega_t} \cos \theta - \frac{v_t}{\omega_t} \cos(\theta + \omega_t \Delta t) \\
        \omega_t \Delta t
        \end{bmatrix} \\
        &= I + 
        \begin{bmatrix}
        0 & 0 & -\frac{v_t}{\omega_t} \cos \theta + \frac{v_t}{\omega_t} \cos(\theta + \omega_t \Delta t) \\
        0 & 0 & -\frac{v_t}{\omega_t} \sin \theta + \frac{v_t}{\omega_t} \sin(\theta + \omega_t \Delta t) \\
        0 & 0 & 0
        \end{bmatrix} \\
        &= 
        \begin{bmatrix}
        1 & 0 & -\frac{v_t}{\omega_t} \cos \theta + \frac{v_t}{\omega_t} \cos(\theta + \omega_t \Delta t) \\
        0 & 1 & -\frac{v_t}{\omega_t} \sin \theta + \frac{v_t}{\omega_t} \sin(\theta + \omega_t \Delta t) \\
        0 & 0 & 1
        \end{bmatrix}
    \end{align*}
    
\end{frame}

        \item Calcular el Jacobiano
        
        \begin{align*}
            \prescript{\text{low}}{}{H^{i}_{t}} &= \frac{\partial h(\bar{\mu}_t)}{\partial \bar{\mu}_t}\\
            &= \frac{1}{q}
            \begin{bmatrix}
                -\sqrt{q} \delta_x & -\sqrt{q} \delta_y & 0 & \sqrt{q} \delta_x & \sqrt{q} \delta_y\\
                \delta_y & -\delta_x & -q & -\delta_y & \delta_x
            \end{bmatrix}
        \end{align*}
    \end{itemize}

\end{frame}

                0 & 0 & 0 & 0 \cdots 0 & 1 & 0 & 0 \cdots 0 \\
                0 & 0 & 0 & 0 \cdots 0 & 0 & 1 & 0 \cdots 0
                \CodeAfter
                \UnderBrace{5-4}{5-4}{2(j-1)}

\begin{frame}
    \frametitle{Algoritmo de Filtro de Kalman Extendido}
    \note{Información extraída de https://www.youtube.com/watch?v=X30sEgIws0g}
    \note{Información extraída de https://www.ipb.uni-bonn.de/html/teaching/photo12-2021/2021-pho2-16-ekf-slam.pptx.pdf}
    
    \begin{algorithmic}[1]
    \Procedure{ExtendedKalmanFilter}{$\mu_{t-1}, \covariance_{t-1}, \controlCommand_{t}, \observation_{t}$}
        \State $\overline{\mu}_{t} = \motionModelFunction{\controlCommand_{t}, \mu_{t-1}}$
        \State $\overline{\covariance}_{t} = \motionModelJacobian_{t} \covariance_{t-1} \motionModelJacobian_{t}^{\top}+\motionParametersCovariance_{t}$
        \Statex
        \State $\alert{\kalmanGain_{t} = \overline{\covariance}_{t} \observationModelJacobian_{t}^{\top} (\observationModelJacobian_{t} \overline{\covariance}_{t}  \observationModelJacobian_{t}^{\top} + \observationModelCovariance_{t})^{-1}}$
        \State $\alert{\mu_{t} = \overline{\mu}_{t} + \kalmanGain_{t} (\observation_{t} - \observationModelFunction{\overline{\mu}_{t}})}$
        \State $\alert{\covariance_{t} =  (I - \kalmanGain_{t} \observationModelJacobian_{t}) \overline{\covariance}_{t}}$
        \State \Return $\mu_{t}, \covariance_{t}$
    \EndProcedure
    \end{algorithmic}
\end{frame}

\begin{frame}
    \frametitle{EKF SLAM - Corrección (1/2)}
    \note{Información extraída de https://www.youtube.com/watch?v=X30sEgIws0g}
    \note{Información extraída de https://www.ipb.uni-bonn.de/html/teaching/photo12-2021/2021-pho2-16-ekf-slam.pptx.pdf}

    \begin{algorithmic}[1]
        \Procedure{EKF\_SLAM\_Correction}{}
        \State $Q_t = \begin{bmatrix} \sigma_r^2 & 0 \\ 0 & \sigma_\theta^2 \end{bmatrix}$ \Comment{Covarianza del ruido de medición}
        \For {todas las features observadas $\observation_i^t = (r_i^t, \phi_i^t)^{\top}$}
            \State $j = c_i^t$ \Comment{asociación de datos (de algún modo)}
            \If{landmark $j$ nunca antes visto}
                \State $\begin{bmatrix} \bar{\mu}_{j,x} \\ \bar{\mu}_{j,y} \end{bmatrix} = 
                       \begin{bmatrix} \bar{\mu}_{t,x} \\ \bar{\mu}_{t,y} \end{bmatrix} + 
                       \begin{bmatrix} r_i^t \cos(\phi_i^t + \bar{\mu}_{t,\theta}) \\ r_i^t \sin(\phi_i^t + \bar{\mu}_{t,\theta}) \end{bmatrix}$
                \Comment{Inicializar la posición del nuevo landmark}
            \EndIf
            \State $\delta = \begin{bmatrix} \delta_x \\ \delta_y \end{bmatrix} = 
                   \begin{bmatrix} \bar{\mu}_{j,x} - \bar{\mu}_{t,x} \\ \bar{\mu}_{j,y} - \bar{\mu}_{t,y} \end{bmatrix}$
            \State $q = \delta^T \delta$
            \State $\hat{z}_i^t = \begin{bmatrix} \sqrt{q} \\ \operatorname{atan2}(\delta_y, \delta_x) - \bar{\mu}_{t,\theta} \end{bmatrix}$
                   \Comment{Medición predicha}
        \algstore{ekf_slam_algorithm_break} % guarda el estado del argoritmo para que este pueda continuar en otro lugar
    \end{algorithmic}
\end{frame}

    \begin{itemize}
        \item Actualizar todas las mediciones en un único paso requiere una sola actualización completa del belief
        \item Siempre normalizar las componentes angulares al intervalo $[-\pi,\pi]$. De lo contrario, esto puede arruinar los Jacobianos.
        \item Puede no ser necesario crear las matrices $F$ explícitamente \note{se pueden manipular directamente los elementos de la matriz de covarianza}
    \end{itemize}

    \note{Tal como describimos el pseudocódigo, necesitamos una actualización completa del belief por cada landmark observado de forma independiente. En la práctica, podemos hacer una única actualización completa del belief usando todos los landmarks observados actualmente para acelerar el cómputo.}

    \frametitle{Correlaciones en EKF SLAM}
    \note{Información extraída de https://www.youtube.com/watch?v=X30sEgIws0g}
    \note{Información extraída de https://www.ipb.uni-bonn.de/html/teaching/photo12-2021/2021-pho2-16-ekf-slam.pptx.pdf}

    \begin{itemize}
        \item En el límite, las estimaciones de los landmarks quedan completamente correlacionadas
    \end{itemize}

    \begin{center}
        \includegraphics[width=0.45\textwidth]{../images/ekf_slam/ekf_slam_correlations.pdf}
    \end{center}

    \scriptsize

    \begin{itemize}
        \item A largo plazo todas las estimaciones quedan completamente correlacionadas. Todas las posiciones de los landmarks están completa (y fuertemente) correlacionadas entre sí (líneas gruesas) y la pose está correlacionada con los landmarks de manera más débil (líneas finas).
        \item La matriz de covarianza representa la incertidumbre y la correlación entre las distintas variables del estado (p.ej., la pose del robot y las posiciones de los landmarks). La \textbf{matriz de correlación} es la matriz de covarianza normalizada de la estimación a posteriori.
        \begin{itemize}
            \item Diagonal: varianza (incertidumbre) de cada variable.
            \item Fuera de la diagonal: correlación entre variables (cómo los errores en una afectan a otra).
        \end{itemize}
    \end{itemize}

    \note{La matriz de correlación simplemente divide la covarianza de las dos variables por el producto de sus desvíos estándar.}
    
\end{frame}

    \note{La línea blanca (valores cero) en la tercera fila y columna de la matriz de correlación proviene de la orientación. Esto significa que la orientación del robot no está correlacionada con las posiciones de los landmarks.}

    \note{Notar que la matriz de correlación se ve como un tablero de ajedrez. Esto se debe a que, en el mapa, cada landmark tiene una covarianza que es aproximadamente un círculo, por lo que en los bloques 2x2 de covarianza de los landmarks de la diagonal principal de la matriz de correlación, los elementos fuera de la diagonal de dicho bloque 2x2 son cercanos a cero.
    Por otro lado, para los elementos fuera de la diagonal de la matriz de correlación: si pudiera fijar la componente x de la posición de un landmark, podría reducir drásticamente la incertidumbre sobre la posición x de todos los demás landmarks también, porque a través de la estimación esos landmarks quedaron conectados de forma prácticamente rígida; si desplazo un landmark con alta incertidumbre a incertidumbre cero, todo el mapa se desplaza también. Por lo tanto, si conozco la posición x precisa de un landmark conoceré también las posiciones x de todos los demás landmarks, pero no aprendo nada sobre y.}

\end{frame}

\begin{frame}
    \frametitle{Resumen de EKF SLAM}
    \note{Información extraída de https://www.youtube.com/watch?v=X30sEgIws0g}
    \note{Información extraída de https://www.ipb.uni-bonn.de/html/teaching/photo12-2021/2021-pho2-16-ekf-slam.pptx.pdf}

    \begin{itemize}
    \item Primer enfoque probabilístico de SLAM, usando EKF
    \item Prueba de convergencia para el caso lineal Gaussiano
    \item Puede diverger si las no linealidades son grandes
    \item Cuanto menor el ruido, mejor
    \item Solo estimaciones unimodales (Gaussianas) \note{EKF no puede manejar incertidumbre multimodal. P.ej.: no puedo decir ``el landmark está aquí o allá, pero no sé exactamente dónde''. ¡El filtro de partículas sí puede!}
    \item Exitoso en escenas de mediana escala
    \item Usado para estimaciones de corto plazo (VO)
    \item Existen aproximaciones para reducir la complejidad computacional
    
    \end{itemize}
\end{frame}
