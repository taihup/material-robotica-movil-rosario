\begin{frame}
  \frametitle{Reordenando la Ecuación DLT}

  \begin{equation*}
    \vec{x}_{i} =
    \begin{bmatrix}
      x_{i} \\
      y_{i} \\
      1
    \end{bmatrix} =
    \begin{bmatrix}
      u_{i} \\
      v_{i} \\
      w_{i}
    \end{bmatrix} = 
    \begin{bmatrix}
      \vec{A}^{\top}\vec{X}_{i} \\
      \vec{B}^{\top}\vec{X}_{i} \\
      \vec{C}^{\top}\vec{X}_{i}
    \end{bmatrix}
  \end{equation*}

Implica

  \begin{equation*}
    x_{i} = \dfrac{u_{i}}{w_{i}} = \dfrac{\vec{A}^{\top}\vec{X}_{i}}{\vec{C}^{\top}\vec{X}_{i}}
    \quad \quad
    y_{i} = \dfrac{v_{i}}{w_{i}} = \dfrac{\vec{B}^{\top}\vec{X}_{i}}{\vec{C}^{\top}\vec{X}_{i}}
  \end{equation*}

\end{frame}

\begin{frame}
  \frametitle{Reordenando la Ecuación DLT}

  \begin{align*}
    x_{i} &= \dfrac{\vec{A}^{\top}\vec{X}_{i}}{\vec{C}^{\top}\vec{X}_{i}} \implies x_{i} \vec{C}^{\top}\vec{X}_{i} - \vec{A}^{\top}\vec{X}_{i} = 0\\
    y_{i} &= \dfrac{\vec{B}^{\top}\vec{X}_{i}}{\vec{C}^{\top}\vec{X}_{i}} \implies y_{i} \vec{C}^{\top}\vec{X}_{i} - \vec{B}^{\top}\vec{X}_{i} = 0
  \end{align*}

  Lleva a un sistema de ecuaciones, que es lineal en los parámetros A, B y C

  \begin{equation*}
    \begin{matrix}
      -\vec{X}_{i}^{\top}\vec{A}& &+x_{i}\vec{X}_{i}^{\top}\vec{C} = 0 \\
      &-\vec{X}_{i}^{\top}\vec{B} &+y_{i}\vec{X}_{i}^{\top}\vec{C} = 0
    \end{matrix}  
  \end{equation*}

\end{frame}

\begin{frame}
  \frametitle{Resolviendo el Sistema Lineal (Sistema Homogéneo => SVD)}
  \begin{itemize}
    \item Resolver un sistema de ecuaciones lineales de la forma $\vec{A}\vec{x} = \vec{0}$ es equivalente a encontrar el espacio nulo de $\vec{A}$
    \item Por lo tanto, podemos aplicar la Descomposición en Valores Singulares (SVD) para resolver $\vec{M}\vec{p} = \vec{0}$
    \item Elegir $\vec{p}$ como el vector singular correspondiente al valor singular 0
  \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Observaciones Redundantes}
    \begin{itemize}
        \item En caso de observaciones redundantes, tendremos contradicciones ($\vec{M}\mathbf{p} \neq \mathbf{0}$):
        %
        \begin{equation*}
            \mathbf{M} \mathbf{p} = \mathbf{w}
        \end{equation*}
        %
        \item Encontrar $\mathbf{p}$ tal que minimice
        %
        \begin{equation*}
            \Omega = \mathbf{w}^{\top} \mathbf{w}
        \end{equation*}
        \begin{align*}
            \implies \mathbf{\hat{p}} &= \underset{\mathbf{p}}{\arg\min} \ \mathbf{w}^{\top} \mathbf{w}\\
            &= \underset{\mathbf{p}}{\arg\min} \ \mathbf{p}^{\top} \mathbf{M}^{\top} \mathbf{M} \mathbf{p}
        \end{align*}
        \begin{equation*}
          \text{con} \lVert\mathbf{P}\rVert_2 = \sum_{i,j} p_{ij}^2 = \lVert\mathbf{p}\rVert = 1
        \end{equation*}
    \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Discussion DLT}
    \begin{itemize}
        \item We realize $\projectionMatrix \rightleftharpoons (\intrinsicMatrix, \rotation, \translation)$ \textbf{both ways} (Projection and Decomposition).
        %\item We are \textbf{free to choose the sign} of the camera constant ($c$).
        \item Solution is \textbf{unstable} if the control points lie \textbf{approximately on a plane}.
        \item Solution is \textbf{statistically not optimal} (no uncertainties of point coordinates are considered).
    \end{itemize}
\end{frame}

\begin{frame}{Resumen}
    \begin{itemize}
        \item La Transformación Lineal Directa estima los parámetros \textbf{intrínsecos} ($\intrinsicMatrix$) y \textbf{extrínsecos} ($\rotation, \translation$) de una cámara.
        \item Calcula los parámetros para el \textbf{mapeo de la cámara no calibrada}.
        \item Requiere al menos \textbf{6 puntos de control conocidos en 3D}.
        \item \textbf{Solución directa} (no se necesita estimación inicial).
    \end{itemize}
\end{frame}
